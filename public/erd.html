erDiagram
    syn_docrequest {
        uniqueidentifier syn_docrequestid PK
        nvarchar syn_title
        nvarchar syn_description
        choice syn_requesttype "Legal | Contract"
        choice syn_requeststatus "Open | In Discussion | In Review | Closed"
        choice syn_priority "Low | Medium | High"
        nvarchar syn_requestedby
        nvarchar syn_assignedto "optional"
        datetime syn_closedat "optional"
        lookup syn_finalversionid "FK → syn_docversion"
        datetime createdon "system"
        datetime modifiedon "system"
    }

    syn_discussion {
        uniqueidentifier syn_discussionid PK
        nvarchar syn_name "auto-number"
        uniqueidentifier syn_docrequestid FK
        uniqueidentifier syn_docversionid "FK optional"
        nvarchar syn_userid
        nvarchar syn_username
        ntext syn_message
        datetime syn_timestamp
        datetime createdon "system"
        datetime modifiedon "system"
    }

    syn_docversion {
        uniqueidentifier syn_docversionid PK
        uniqueidentifier syn_docrequestid FK
        nvarchar syn_title
        int syn_versionnumber
        nvarchar syn_documenturl
        nvarchar syn_description "optional"
        nvarchar syn_uploadedby
        datetime syn_uploadedat
        twooption syn_signedoff "Yes/No"
        nvarchar syn_signedoffby "optional"
        datetime syn_signedoffat "optional"
        datetime createdon "system"
        datetime modifiedon "system"
    }

    syn_docrequest ||--o{ syn_discussion : "has many"
    syn_docrequest ||--o{ syn_docversion : "has many"
    syn_docversion ||--o{ syn_discussion : "version comments"
    syn_docrequest |o--o| syn_docversion : "final version"
